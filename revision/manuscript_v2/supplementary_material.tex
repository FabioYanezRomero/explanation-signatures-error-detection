\section{Multidimensional Analysis Metrics} \label{sec:metrics}

\setlength{\tabcolsep}{3pt}
\renewcommand{\arraystretch}{1.15}
\begin{table}[H]
\centering
\tiny
\caption{Multi-Dimensional Explainability Evaluation Framework: Complete Metrics Overview}
\label{tab:all_dimensions_metrics}
{\rowcolors{3}{gray!5}{white}
\begin{tabular}{@{}p{3.0cm}p{11.0cm}@{}}
\toprule
\textbf{Metric} & \textbf{Definition and Operational Role} \\
\midrule
\multicolumn{2}{@{}l}{\textbf{Dimension 1: Error Detection Capacity---Insertion-Deletion AUC}} \\
\midrule
\textbf{Deletion AUC} & AUC of confidence drops as top-ranked features are progressively removed: $\text{AUC}_{\text{del}} = \int_{0}^{1} \text{conf}(f(x - \text{features}_k)) \, dk$. Low values indicate feature necessity; high values indicate features are not critical. \\
\textbf{Insertion AUC} & AUC of confidence recovery when reintroducing features from neutral baseline: $\text{AUC}_{\text{ins}} = \int_{0}^{1} \text{conf}(f(x_{\text{neutral}} + \text{features}_k)) \, dk$. High values indicate collective feature sufficiency for model behaviour. \\
\midrule[\heavyrulewidth]
\multicolumn{2}{@{}l}{\textbf{Dimension 2: Feature Ranking Stability}} \\
\midrule
\textbf{Maskout Drop Progression} & Cumulative confidence drops $\{\Delta_1, \Delta_2, \ldots, \Delta_K\}$ as features are progressively masked: $\Delta_k = \text{conf}(f(x)) - \text{conf}(f(x - \text{features}_{1:k}))$. Indicates monotonic degradation consistency in importance rankings. \\
\textbf{Sufficiency Drop} & Terminal confidence drop after masking all top-$K$ features: $\Delta_K = \text{conf}(f(x)) - \text{conf}(f(x - \text{features}_{1:K}))$. Quantifies explanation completeness via the fraction of confidence attributable to identified features. \\
\textbf{Importance Concentration} & Cumulative importance fraction in top-$N$ features: $C(N) = \frac{\sum_{i=1}^{N} \text{importance}_i}{\sum_{i=1}^{K} \text{importance}_i}$. Reveals hierarchical sharpness versus distributed importance in explanations. \\
\midrule[\heavyrulewidth]
\multicolumn{2}{@{}l}{\textbf{Dimension 3: Explanation Consistency Analysis}} \\
\midrule
\textbf{Origin Contrastivity} & Margin on unmodified input: $\text{conf}(\hat{y}) - \text{conf}(\hat{y}_2)$ where $\hat{y}$ is predicted class and $\hat{y}_2$ is runner-up. Establishes baseline discrimination, stratified by correctness and class. \\
\textbf{Masked Contrastivity} & Margin after zeroing all features except explainer-identified important ones: $\text{conf}_{\text{masked}}(\hat{y}) - \text{conf}_{\text{masked}}(\hat{y}_2)$. A high margin indicates that the identified features sufficiently support the prediction. \\
\textbf{Maskout Contrastivity} & Margin after zeroing top-ranked features while keeping others: $\text{conf}_{\text{maskout}}(\hat{y}) - \text{conf}_{\text{maskout}}(\hat{y}_2)$. Margin collapse indicates necessity; comparison across correct/incorrect predictions reveals differential feature function. \\
\midrule[\heavyrulewidth]
\multicolumn{2}{@{}l}{\textbf{Dimension 4: Explanation Faithfulness via Fidelity Asymmetry}} \\
\midrule
\textbf{Fidelity+ (Insertion)} & Confidence increase when adding top-ranked features to neutral baseline: \(M_+ = \text{conf}(f(x + \text{features})) - \text{conf}(f(x_\text{neutral}))\). Measures collective decision reproduction. \\
\textbf{Fidelity- (Deletion)} & Confidence decrease when removing top-ranked features: \(M_- = \text{conf}(f(x)) - \text{conf}(f(x - \text{features}))\). Measures performance degradation from feature absence. \\
\textbf{Asymmetry} & Comparative balance metric: \(A = (M_- - M_+) / (|M_-| + |M_+|)\). Range \([-1, +1]\) quantifies deletion versus insertion effect magnitude without prescriptive interpretation. \\
\bottomrule
\end{tabular}
}
\end{table}
